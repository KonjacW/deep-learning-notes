```mermaid
flowchart TD
    A([开始: 定义任务与真实业务目标])

    A --> B["定义评估体系<br/>1. 设定单一数字指标 single-number metric<br/>2. 区分 optimizing metric 与 satisficing metrics<br/>3. 确保指标能反映真实产品体验"]
    B --> C{"指标是否真正对应业务目标?"}

    C -- 否 --> C1["问题类型: Metric 设计错误<br/>表现: 指标变好但真实体验变差<br/>策略:<br/>1. 重定义主指标<br/>2. 对严重错误加权 Weighted Error<br/>3. 重新明确约束条件"]
    C1 --> B

    C -- 是 --> D["设计数据集<br/>1. Train 用于学习参数<br/>2. Dev 用于选方向/选模型<br/>3. Test 用于最终评估<br/>4. Dev/Test 必须同分布且贴近真实场景<br/>5. 训练集尽量大, Dev/Test 只需足够评估"]

    D --> E{"Train 与 Dev/Test 是否同分布?"}

    E -- 是 --> F["常规数据划分流程<br/>直接进入基线系统构建"]
    E -- 否 --> G["分布不匹配设定<br/>1. 允许 Train 更宽松、更大<br/>2. Dev/Test 必须贴近目标场景<br/>3. 额外保留 Training-dev set<br/>   来源于 Train 分布但不参与训练"]

    F --> H["快速搭建第一个可运行系统 / baseline<br/>目的: 尽快看到真实错误, 不先盲目深挖"]
    G --> H

    H --> I{"选择建模路径"}

    I -- "常规单任务" --> I1["常规监督学习流程"]
    I -- "目标数据少, 有相关大任务数据" --> I2["迁移学习 Transfer Learning<br/>1. 在任务A预训练 pre-training<br/>2. 替换输出层<br/>3. 视数据量选择:<br/>   - 只训最后几层<br/>   - 整网 fine-tuning"]
    I -- "多个相关任务共享输入" --> I3["多任务学习 Multi-task Learning<br/>1. 一个网络并行学多个任务<br/>2. 适合共享表示、多标签输出场景<br/>3. 前提: 任务相关 + 每任务有数据 + 网络容量足够"]
    I -- "想做端到端" --> I4{"端到端数据是否足够多?"}

    I4 -- 否 --> I5["问题类型: End-to-End 不适配<br/>策略:<br/>1. 改为分阶段流水线<br/>2. 利用人类先验与中间模块<br/>例: 检测->识别, 感知->规划->控制"]
    I5 --> I
    I4 -- 是 --> I6["端到端学习 End-to-End<br/>直接学习 x -> y<br/>优点: 少手工模块<br/>风险: 映射复杂时极度吃数据"]

    I1 --> J["训练模型"]
    I2 --> J
    I3 --> J
    I6 --> J

    J --> K["得到关键误差指标<br/>Human-level / Train / Training-dev(若有) / Dev / Test"]
    K --> L{"当前数据设定是否存在 Train-Dev/Test 分布差异?"}

    %% 同分布诊断
    L -- 否 --> M["同分布诊断<br/>1. Avoidable Bias ≈ Train Error - Human-level Error<br/>2. Variance ≈ Dev Error - Train Error<br/>3. Dev overfit ≈ Test Error - Dev Error"]

    M --> N{"Train Error 是否远高于 Human-level Error?"}
    N -- 是 --> N1["问题类型: 高可避免偏差 High Avoidable Bias<br/>含义: 训练集都没拟合好"]
    N1 --> N2["策略:<br/>1. 更大网络<br/>2. 训练更久<br/>3. 更好的优化算法: Momentum / RMSprop / Adam<br/>4. 更好网络结构<br/>5. 调超参数"]
    N2 --> J

    N -- 否 --> O{"Dev Error 是否明显高于 Train Error?"}
    O -- 是 --> O1["问题类型: 高方差 High Variance<br/>含义: 训练集学得还行, 泛化差"]
    O1 --> O2["策略:<br/>1. 收集更多数据<br/>2. 正则化<br/>3. Dropout<br/>4. 数据增强<br/>5. 调结构与超参数"]
    O2 --> J

    O -- 否 --> P{"Test Error 是否明显高于 Dev Error?"}
    P -- 是 --> P1["问题类型: 对 Dev 过拟合<br/>含义: 反复根据 Dev 调参, 泛化估计偏乐观"]
    P1 --> P2["策略:<br/>1. 减少围绕 Dev 的过度试验<br/>2. 增强独立 Test 检查<br/>3. 必要时重建 Dev/Test"]
    P2 --> D

    P -- 否 --> Q{"Dev/Test 表现是否仍不满足业务要求?"}

    %% 异分布诊断
    L -- 是 --> R["异分布诊断<br/>1. Avoidable Bias = Train - Human-level<br/>2. Variance = Training-dev - Train<br/>3. Data Mismatch = Dev - Training-dev<br/>4. Dev overfit = Test - Dev"]

    R --> S{"Train Error 是否远高于 Human-level Error?"}
    S -- 是 --> S1["问题类型: 高可避免偏差"]
    S1 --> S2["策略:<br/>1. 增强拟合能力<br/>2. 更大模型/更久训练/更好优化器/更好结构/调参"]
    S2 --> J

    S -- 否 --> T{"Training-dev Error 是否明显高于 Train Error?"}
    T -- 是 --> T1["问题类型: 高方差"]
    T1 --> T2["策略:<br/>1. 更多训练数据<br/>2. 正则化 / Dropout<br/>3. 数据增强<br/>4. 结构与超参数调整"]
    T2 --> J

    T -- 否 --> U{"Dev Error 是否明显高于 Training-dev Error?"}
    U -- 是 --> U1["问题类型: Data Mismatch 分布不匹配"]
    U1 --> U2["先做针对 Dev 错误的人工分析<br/>理解 Train 与 Dev 差异来源:<br/>噪声 / 清晰度 / 背景 / 场景语义 / 子任务频率"]
    U2 --> U3["策略A: 收集更像 Dev/Test 的训练数据"]
    U2 --> U4["策略B: 人工数据合成 Artificial Data Synthesis<br/>例: 干净语音 + 车内噪声"]
    U4 --> U5{"合成数据是否覆盖足够多样?"}
    U5 -- 否 --> U6["风险: 对狭窄合成规则过拟合<br/>策略: 扩大噪声/场景/组合多样性"]
    U6 --> U4
    U5 -- 是 --> J
    U3 --> J

    U -- 否 --> V{"Test Error 是否明显高于 Dev Error?"}
    V -- 是 --> V1["问题类型: 对 Dev 过拟合"]
    V1 --> V2["策略:<br/>1. 强化独立 Test 评估<br/>2. 控制基于 Dev 的过多反复调参<br/>3. 必要时重建数据划分"]
    V2 --> D

    V -- 否 --> Q

    %% Error analysis and label issues
    Q -- 是 --> W["做 Error Analysis 误差分析<br/>1. 从 Dev 收集一批错分样本(如100个)<br/>2. 人工查看<br/>3. 给每个样本打错误类型标签<br/>4. 统计占比与可修复性<br/>5. 按收益上限估计优先级"]
    W --> W1{"哪类错误占比高且可修复?"}

    W1 -- "特定视觉/语音/场景子类错误" --> W2["针对高占比错误定向优化<br/>可选:<br/>1. 补该类数据<br/>2. 数据增强<br/>3. 结构改进<br/>4. 调阈值/损失/训练策略"]
    W2 --> J

    W1 -- "怀疑标签问题" --> X{"标签错误发生在哪?"}

    X -- "Train" --> X1{"训练集标签错误是随机还是系统性?"}
    X1 -- 随机 --> X2["结论: 大数据下常可暂缓处理<br/>原因: 深度学习对随机标注错较鲁棒<br/>回流: 继续优先处理更高收益问题"]
    X2 --> W

    X1 -- 系统性 --> X3["问题类型: 系统性训练标签错误<br/>例: 某类样本总被错标"]
    X3 --> X4["策略:<br/>1. 定位错误模式<br/>2. 定向清洗/重标该子类<br/>3. 必要时修订采标流程"]
    X4 --> J

    X -- "Dev/Test" --> Y["问题类型: Dev/Test 标签错误影响模型选择与最终评估"]
    Y --> Y1{"标签错误贡献是否已接近主要误差来源?"}
    Y1 -- 否 --> Y2["影响暂时较小<br/>可先继续处理更大的真实模型问题"]
    Y2 --> W

    Y1 -- 是 --> Y3["策略:<br/>1. 优先清理 Dev/Test<br/>2. Dev 和 Test 一起修, 保持同分布<br/>3. 理想上同时检查预测对/错样本, 避免评估偏置"]
    Y3 --> D

    W1 -- "错误分散, 暂无明显主因" --> W3["结论: 当前收益最大的方向不明确<br/>策略:<br/>1. 保持 baseline<br/>2. 补充分析维度<br/>3. 继续围绕业务目标细化 metric 或数据"]
    W3 --> B

    %% final success
    Q -- 否 --> Z{"是否同时满足:<br/>1. Dev/Test 稳定<br/>2. 指标符合业务目标<br/>3. 无明显 Dev 过拟合?"}
    Z -- 否 --> W
    Z -- 是 --> AA([完成: 进入最终测试/部署评估])

    %% orthogonalization notes
    AA --> AB["正交化原则回顾<br/>1. 先解决训练集拟合 -> 偏差<br/>2. 再解决 Dev 泛化 -> 方差<br/>3. 再检查 Test 稳定 -> 防 Dev 过拟合<br/>4. 再确认指标与分布 -> 对准真实业务"]
```
